% \partienexus{numéro}{titre}{durée conseillée}
\newcommand{\partienexus}[3]{%
  % Partie 1 sans saut forcé ; toute Partie 2 et suivantes sur une nouvelle page.
  \ifnum#1>1 \clearpage \fi
  \needspace{13\baselineskip}%
  \par\vspace{0.8\baselineskip}%
  \noindent\begin{tcolorbox}[enhanced,colback=navy,colframe=navy,boxrule=0pt,
      left=4mm,right=4mm,top=2.4mm,bottom=2.4mm,arc=0.8mm,

               at ([xshift=-4mm]frame.east) {#1};}]

    {\color{white}\bfseries\large #2}%
    \ifx\relax#3\relax\else\\[2pt]{\color{ivoire}\footnotesize Temps conseillé : #3}\fi
  \end{tcolorbox}\par\vspace{0.4\baselineskip}}

% ── cartouche de couverture ────────────────────────────────────────────────────
\newcommand{\cartouche}[2]{%
  \begin{tcolorbox}[enhanced,colback=ivoire,colframe=griscadre,boxrule=0.4pt,
      left=3mm,right=3mm,top=2mm,bottom=2mm,arc=0.6mm,width=\linewidth,
      % 17 mm : « Baccalauréat complet en une session » tient sur deux lignes, et
      % débordait du cadre quand la hauteur était calée sur une seule.
      valign=center,height=17mm]
    % Pas de césure dans un cartouche : « ses-sion » coupé en deux se lit mal et donne
    % au document un air de formulaire mal composé.
    \hyphenpenalty=10000\exhyphenpenalty=10000\relax
    {\scriptsize\color{gristech}\MakeUppercase{#1}}\\[1pt]{\small\bfseries\color{navy}#2}
  \end{tcolorbox}}

% ── extrait littéraire ─────────────────────────────────────────────────────────
% \extrait{titre}{texte}{référence}
\newcommand{\extrait}[3]{%
  \needspace{8\baselineskip}%
  \par\vspace{0.5\baselineskip}%
  {\footnotesize\bfseries\color{navy}\MakeUppercase{#1}}\par\vspace{0.2\baselineskip}%
  % width et boxsep explicites : sans eux la boîte débordait du bloc de texte à droite.
  \begin{tcolorbox}[enhanced,blank,width=\linewidth,boxsep=0pt,
      borderline west={1.2pt}{0pt}{griscadre},
      left=4mm,right=2mm,top=1mm,bottom=1mm,before skip=2pt,after skip=2pt]
    {\serifdelecture\fontsize{11}{16}\selectfont
     \setlength{\parskip}{0.5\baselineskip}\setlength{\parindent}{0pt}#2\par}
  \end{tcolorbox}%
  \par\vspace{0.1\baselineskip}{\footnotesize\color{gristech}#3}\par
  \vspace{0.3\baselineskip}}

% ── documents d'appui : figure et tableau ──────────────────────────────────────
% Une figure n'est jamais réduite au point d'être illisible : sa hauteur est plafonnée,
% jamais sa largeur au-delà du rapport. Le titre la précède, la provenance la suit, plus
% discrète que le contenu. Le tout est réservé d'un bloc avec la question qui l'exploite.
% \figurenexus{titre}{chemin}{provenance}
\newcommand{\figurenexus}[3]{%
  \par\vspace{0.4\baselineskip}%
  {\footnotesize\bfseries\color{navy}#1}\par\vspace{0.25\baselineskip}%
  \noindent\begin{tcolorbox}[enhanced,colback=white,colframe=griscadre,boxrule=0.4pt,
      left=2mm,right=2mm,top=2mm,bottom=1.5mm,arc=0.6mm,width=\linewidth,halign=center]
    \includegraphics[width=\linewidth,height=70mm,keepaspectratio]{#2}
  \end{tcolorbox}%
  \par\vspace{0.1\baselineskip}{\scriptsize\color{gristech}#3}\par
  \vspace{0.3\baselineskip}}

% ── repère de profil ───────────────────────────────────────────────────────────
% \reperprofil{étapes en or}{libellé}. Trois traits : le premier pour les épreuves
% anticipées, le second pour les terminales, les deux pour un bac en une session. La
% couleur ne porte rien seule — le libellé est écrit à côté.
\newcommand{\reperprofil}[2]{%
  \begin{tikzpicture}[baseline=0pt]
    \foreach \i in {1,2} {
      \pgfmathparse{\i<=#1 ? "or" : "ivoire!35"}
      \fill[\pgfmathresult] ({(\i-1)*8mm},0) rectangle ({(\i-1)*8mm+6.5mm},1.2mm);
    }
    \node[anchor=north west,text=ivoire,font=\scriptsize] at (0,-1.2mm) {#2};
  \end{tikzpicture}}
